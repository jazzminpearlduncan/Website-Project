library(tidyverse)

  # rules for how to search the folder
  list.files(
    path = "MDSDATA",
    # see: regular expressions
    pattern = ".csv$" ,
    full.names = TRUE
  )


all_years_data <- setNames(
  lapply(all_years_files, read_csv),
  basename(all_years_files))
